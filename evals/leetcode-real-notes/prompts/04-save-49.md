```python
class Solution:
    def groupAnagrams(self, strs: list[str]) -> list[list[str]]:
        record = {}
        for s in strs:
            unique = tuple(sorted(Counter(s).items()))
            if unique in record:
                record[unique].append(s)
            else:
                record[unique] = [s]

        return list(record.values())
```
